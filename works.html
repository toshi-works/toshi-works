                from sklearn.linear_model import LinearRegression

# データの読み込み
data = pd.read_csv('california_housing.csv')

# 重回帰分析の実行
X = data[['MedInc', 'HouseAge', 'AveRooms']]
y = data['MedHouseValue']

model = LinearRegression()
model.fit(X, y)

print(f'決定係数: {model.score(X, y):.4f}')
